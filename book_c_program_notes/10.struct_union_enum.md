## Intoduction to Structure

- This is foundation for object oriented program and user defined datatype.
- The Need for User-Defined Data Types.

```c
void the_data_grouping_problem()
{
    /*
       Managing multiple related variables (like employee records)
       using only primary data types becomes unscalable and disorganized.
    */

    // Employee 1
    int id;
    char name[20];
    int sal;

    // Employee 2
    int id2;
    char name2[20];
    int sal2;

    // ......

    // Employee 10
    int id10;
    char name10[20];
    int sal10;

    // Group the datatype logically. Here `emp` structure and `e1, e2` are structure variable
    emp e1, e2, .... e10;
}
```

### C Data Type Classification Hierarchy

- **datatype**
  - **Primary**
    - `int`
    - `float`
    - `char`
    - `double`
  - **Secondary**
    - **derived**
      - `signed`
      - `unsigned`
      - `short`
      - `long`
      - `pointers`
      - `Arrays`
    - **UDDT (User defined data type)**
      - `struct`
      - `union`
      - `enum`

### Operator used in structure

- `.` (dot operator)
  - Syntax
    - `structure variable`.`structure member`
- `->`(arrow operator)
  - syntax
    - `structure pointer` -> `struture member`
- The two operators are called member selector operator.

### Structure Declaration and Memory Layout

```c
#include <stdio.h>
#include <string.h>

// ==========================================
// 1. Structure Definition & Terminology
// ==========================================
/*
   - 'struct emp' represents the datatype.
   - 'emp' is the struct name.
   - The variables defined inside are called Structure Members (Any thing which is present inside structure).
   - These members are the variables which are used to tell the size info of structure to Compiler.
*/
struct emp {
    int id;           // 2 bytes
    char name[36];    // 36 bytes
    int sal;          // 2 bytes
};                            // Total size: 40 Bytes (Assuming a 16-bit compiler where int is 2 bytes)

void structure_instantiation_and_access() {

    /*
       Declaration Analogy:
       Just as 'int x;' declares variable 'x' of type 'int',
       'struct emp e1;' declares variable 'e1' of type 'struct emp'.
    */
    // emp e1;                // X Invalid: Requires the 'struct' keyword
    struct emp e1;            // (ok) 'e1' is the Structure Variable (Allocates 40 Bytes)

    // ==========================================
    // 2. Member Access Syntax: SV . SM
    //    (Structure Variable . Structure Member)
    // ==========================================

    e1.id = 1;                // Valid integer assignment

    // e1.name = "abc";       // Invalid: Cannot assign string literal directly to a character array after declaration
    strcpy(e1.name, "abc");   // Valid: Use string copy for character arrays

    e1.sal = 5000;            // Valid integer assignment

    /*
      ==========================================
      3. Contiguous Memory Layout for 'e1'
      ==========================================
      Structure members are allocated in sequential memory addresses.

        id        name                                      sal
      ╭──────┬──────────────────────────────────────────┬──────╮
      │  1   │ abc                                      │ 5000 │
      ╰──────┴──────────────────────────────────────────┴──────╯
       500    502                                        538
    */
}
```

## Structure Initialization

```c
// Structure Initialization Methods and Designated Initializers

#include <stdio.h>

// ==========================================
// 1. Employee Structure Definition
// ==========================================
struct emp {
    int id;
    char name[36];
    int sal;
};                                // Total size: 40 Bytes

void structure_initialization_techniques() {
    /*
      Analogy for initialization:
      1. int x; x = 10; (Declaration then assignment)
      2. int x = 10;    (Initialization at declaration)
    */

    // 1. Complete Positional Initialization
    struct emp e1_full = { 1, "abc", 5000 };

    // 2. Partial Initialization (Trailing elements default to 0)
    struct emp e1_partial1 = { 1, "abc" };          // sal -> 0

    // 3. Single Element Initialization (Remaining elements default to 0)
    struct emp e1_partial2 = { 1 };                 // name -> 0, sal -> 0

    // 4. Invalid Positional Skipping
    struct emp e1_invalid = { 1, , 5000 };       // Invalid syntax

    // 5. Designated Initializers (C99 feature for explicit member assignment)
    struct emp e1_designated = { .id = 1, .sal = 5000 };

    // 6. Explicit NULL Initialization for Arrays
    struct emp e2 = { 1, NULL, 5000 };
}

// ==========================================
// 2. Designated Initializers vs Standard Initialization
// ==========================================
struct K {
    int a;
    int b;
    int c;
    int d;
    int e;
    int f;
};

void designated_initializer_advantages() {
    // Goal: Initialize only specific members (a = 111, c = 222, e = 333)

    // Traditional Positional Approach (Requires manually entering zeros for skipped fields)
    struct K k1_standard = { 111, 0, 222, 0, 333, 0 };

    // Designated Initializer Approach (Cleaner, explicitly names members to initialize)
    // Unlisted members (b, d, f) automatically default to 0.
    struct K k1_designated = { .a = 111, .c = 222, .e = 333 };
}
```

### Assigning data at run time

```c
// Structure Input via scanf/fgets and Operator Precedence

#include <stdio.h>

struct emp {
    int id;
    char name[36];
    int sal;
};                                // Total size: 40 Bytes

void structure_input_methods() {
    /*
      Analogies for variable assignment:
      (1)int x = 10;               // Initialization at declaration
      (2)int x; x = 10;            // Declaration followed by assignment
      (3)int x; scanf("%d", &x);   // Dynamic input via scanf
    */

    struct emp e1;

    // 1. Reading Primitive Members via scanf
    printf("enter id, sal\n");
    scanf("%d %d", &e1.id, &e1.sal);

    /*
       Operator Precedence Breakdown: &e1.id
       We are applying two operators:
       - Dot operator (.) for member access
       - Address-of operator (&) for pointer generation

       According to C precedence rules, the dot operator (.) has higher precedence
       than the address-of operator (&).

       Execution Order:
       Step 1: e1.id (Evaluates to the member variable)
       Step 2: &(e1.id) (Generates the address of that specific member)

       Note: e1.&id is INVALID syntax.

       Analogy: 2 + 3 * 4
       Precedence dictates multiplication happens first: 2 + (3 * 4) = 14
    */

    // 2. Reading String Members safely via fgets
    printf("enter name\n");

    // Flush standard input before reading string to clear lingering newlines from scanf
    int c; while ((c = getchar()) != '\n' && c != EOF);

    fgets(e1.name, sizeof(e1.name), stdin);

    // 3. Printing the populated structure
    printf("%d %s %d", e1.id, e1.name, e1.sal);
}
```

### Structure Pointers and the Arrow Operator (->)

```c
#include <stdio.h>

struct emp {
    int id;
    char name[36];
    int sal;
};

void structure_pointer_access() {
    // 1. Initialization and Pointer Declaration
    struct emp e = {1, "abc", 3000};

    // ep is a Structure Pointer / Employee Pointer
    struct emp *ep;

    // 2. Assigning the base address of the structure to the pointer
    ep = &e;

    /*
       Access Syntax: SP -> SM
       (Structure Pointer -> Structure Member)
    */

    // 3. Accessing and Modifying Members
    printf("%d", ep->id);                  // Output: 1

    ep->sal = ep->sal + 2000;              // Updates salary from 3000 to 5000
    printf("%d", ep->sal);                 // Output: 5000

    // 4. Operator Precedence and Array Indexing
    printf("%c", ep->name[1]);             // Output: 'b'

    /*
       Operator Precedence Breakdown for: ep->name[1]

       The arrow operator (->) is evaluated first (labeled ①),
       followed by the array subscript operator [] (labeled ②).

       Evaluation steps:
       Step ①: ep->name        => Evaluates to base address 502
       Step ②: (ep->name)[1]   => Evaluates to *(ep->name + 1)
                               => *(502 + 1) => 'b'

       Precedence Analogy provided:
       5 * 3 / 2
       (1)  (2) (Left-to-right associativity for operators of same precedence)
    */
}

/*
  ==========================================
  Memory Layout and Pointer Visualization
  ==========================================

     ep Pointer
    ╭─────╮
    │ 500 │
    ╰──┬──╯
       │       id       name                                      sal
       │     ╭──────┬──────────────────────────────────────────┬────────╮
       ╰───> │  1   │ abc                                      │~~3000~~│
             │      │                                          │  5000  │
             ╰──────┴──────────────────────────────────────────┴────────╯
              500    502                                        538
*/
```

## Pointer member in structure

```c
// Structure Pointer Members and Double Pointers

#include <stdio.h>

// ==========================================
// 1. Structure with Pointer Members (SPM)
// ==========================================
struct A {
    int i;                          // SM (Structure Member)
    int *ip;                        // SPM (Structure Pointer Member)
};                                  // 4 Bytes (Assuming 16-bit architecture)

void struct_pointer_members() {
    struct A a;                     // SV (Structure Variable)
    struct A *ap;                   // SP (Structure Pointer)
    int x = 888;

    // Direct Member Access
    // a.i => int data
    a.i = 10;

    // a.ip => int address (int *)
    a.ip = &x;

    /*
       Evaluation of *a.ip:
       The dot operator (.) has higher precedence than the dereference operator (*).
       It evaluates as *(a.ip).
       Analogy: (2+3*5) -> multiplication happens first.
    */
    // Output conceptually: 10 100 888 (assuming x is at address 100)
    printf("%d %d %d", a.i, a.ip, *a.ip);

    // Pointer Member Access via Structure Pointer
    ap = &a;

    ap->i => 10
    ap->x // C.E: x is not a member of Struct A
    ap->ip => 100 => int *

    /*
       Evaluation of *ap->ip:
       The arrow operator (->) has higher precedence than (*).
       It evaluates as *(ap->ip) => 888 => int
    */
    int val = *ap->ip;              // Valid

    // Invalid Syntax Forms:
    a.*ip    // X (Invalid)
    ap->*ip  // X (Invalid)
}

/*
  Memory Layout for struct A and pointer 'ap'

     ap       a               x
    ╭───╮    ╭────┬─────╮    ╭─────╮
    │500│ ─> │ 10 │ 100 │ ─> │ 888 │
    ╰───╯    ╰────┴─────╯    ╰─────╯
              500  502        100
               i    ip
*/

// ==========================================
// 2. Double Pointers to Structures
// ==========================================
struct emp {
    int id;
    char name[36];
    int sal;
};

void double_structure_pointers() {
    struct emp e = {1, "abc", 3000};
    struct emp *ep;
    struct emp **epp;               // Double pointer to struct emp

    ep = &e;
    epp = &ep;

    // --- Accessing 'id' via Single Pointer (ep) ---
    ep->id                       // (ok) Valid
    (*ep).id                     // (ok) Valid
    *ep.id                       // X X (Invalid precedence, evaluates as *(ep.id))

    // --- Accessing 'id' via Double Pointer (epp) ---
    epp->id                      // X X (Invalid, epp points to a pointer, not a struct)
    *epp->id                     // X X (Invalid precedence)
    (*epp)->id                   // (ok) Valid (*epp evaluates to ep, then ep->id)
    (**epp).id                   // (ok) Valid (**epp evaluates to e, then e.id)
}

/*
  Memory Layout for Double Pointers

     epp      ep        e (id, name, sal)
    ╭───╮    ╭───╮    ╭───┬─────┬──────╮
    │600│ ─> │500│ ─> │ 1 │ abc │ 3000 │
    ╰───╯    ╰───╯    ╰───┴─────┴──────╯
     600      500      500 502   538
*/
```

### Summary

```c
// Structure Member Access Syntax Summary

/*
  Terminology Mapping:
  SM  = Structure Member
  SPM = Structure Pointer Member

  SV  = Structure Variable
  SP  = Structure Pointer
  SPP = Structure Pointer to Pointer (Double Pointer)
*/

void structure_access_syntax_rules() {

    // ==========================================
    // 1. Single Pointer & Direct Access
    // ==========================================

    // (1)x . y     => x is SV, y is SM
    // Example: e.id

    // (2)x -> y    => x is SP, y is SM
    // Example: ep->id

    // (3)(*x) . y  => x is SP, y is SM
    // Example: (*ep).id
    // Note: Rules (2)and (3)evaluate to the Same result.

    // ==========================================
    // 2. Dereferencing Pointer Members (SPM)
    // ==========================================

    // (4)*x . y    => x is SV, y is SM(SPM)
    // Example: *a.ip

    // (5) *x -> y   => x is SP, y is SPM
    // Example: *ap->ip

    // ==========================================
    // 3. Double Pointer Access (SPP)
    // ==========================================

    // (6) (*x) -> y => x is SPP, y is SM
    // Example: (*epp)->id

    // ⑦ (**x) . y => x is SPP, y is SPM
    // Example: (**epp).id
    // (Note: The board transcription lists 'y' as SPM here, though 'id' is a standard SM)

    // ==========================================
    // 4. Lexical Parsing (Maximal Munch Rule)
    // ==========================================

    /*
       Parsing ambiguity resolution:
       x -> y <- z   (ok)

       The compiler parses the consecutive operators '<-' by breaking them down
       into the less-than operator '<' and the unary minus '-':

       Evaluates as:
       x -> y < -z
    */
}
```

## Operations between two structure variable

- Assignment operation is the only possible operation between 2 structure
  variable

```c
// Structure Operations: Assignment, Arithmetic, and Comparison

struct emp {
    int id;
    char name[36];
    int sal;
};

void structure_operations() {
    // 1. Initialization and Declaration
    struct emp e1 = {1, "abc", 3000};
    struct emp e2, e3;

    // ==========================================
    // Valid Structure Operations
    // ==========================================

    // Direct structure assignment (performs a member-wise block copy)
    e2 = e1;                                // (ok) Valid

    // Modifying individual members independently
    e2.id = 2;                              // Modifies e2 without affecting e1

    // Relational operations on specific members
    if (e1.id > e2.id) { }                  // (ok) Valid

    // Arithmetic operations on specific members
    int total_sal = e1.sal + e2.sal;        // (ok) Valid


    // ==========================================
    // Invalid Structure Operations (Compile Errors)
    // ==========================================

    // Direct arithmetic on entire structures
    e3 = e1 + e2;                        // X Invalid

    // Direct relational comparison of entire structures
    if (e1 == e2) { }                    // X Invalid

    // Direct increment/decrement of entire structures
    ++e1;                                // X Invalid
}

/*
  ==========================================
  Memory Layout Post-Assignment and Modification
  ==========================================
  When e2 = e1 executes, the contents of e1 are copied to e2's separate memory block.
  Subsequent modifications to e2.id only affect e2's memory.

     e1  ╭───────┬──────────────────────────────────────────┬──────╮
         │   1   │ abc                                      │ 3000 │
         ╰───────┴──────────────────────────────────────────┴──────╯
          id (500) name (502)                                sal (538)

     e2  ╭───────┬──────────────────────────────────────────┬──────╮
         │ ~~1~~ │                                          │      │
         │   2   │ abc                                      │ 3000 │
         ╰───────┴──────────────────────────────────────────┴──────╯
          id (600) name (602)                                sal (638)
*/
```

## Containership (stuct inside the struct)

- Declaring one structure variable as a member to another structure is called
  containership or composition.

```c
// Containership, Memory Layout, and Nested Pointer Access

#include <stdio.h>

// ==========================================
// 1. Basic Composition Definition
// ==========================================
struct A {
    int i;
    int j;
};                                  // 4 Bytes (Assuming 2-byte integers)

struct B {                          // Containership / Composition
    struct A a;
    int x;
};                                  // 6 Bytes total

void nested_structure_access() {
    struct B b;
    struct B *pb;

    // Direct Access via dot operator (.)
    // b.a => struct A type
    // b.x => int type
    b.a.i = 10;
    b.a.j = 20;
    b.x = 30;

    // Pointer Access via arrow operator (->)
    pb = &b;

    // Accessing nested members through a pointer:
    pb->a.i // => 10
    pb->a.j // => 20
    pb->x   // => 30

    /*
       Operator Equivalency and Precedence:
       x->y is functionally equivalent to (*x).y
       Therefore, pb->a.i evaluates the same as (*pb).a.i
    */
}

/*
  Memory Layout for Nested Structure 'b' accessed by pointer 'pb'

     pb           a (struct A)           x
    ╭───╮        ╭──────┬──────╮      ╭──────╮
    │500│ ─────> │  10  │  20  │      │  30  │
    ╰───╯        ╰──────┴──────╯      ╰──────╯
                  500    502           504
                   i      j
*/
```

- Nested Structure Pointers and Deep Member Access

```c
#include <stdio.h>

// ==========================================
// 1. Structure with a Pointer to Another Structure
// ==========================================
struct A {
    int x;
    int y;
};                                   // 4 Bytes

struct B {
    struct A *pa;                    // Structure Pointer Member (Points to struct A)
    int i;
};                                   // 4 Bytes (Assuming 16-bit pointers)

void nested_pointer_access() {
    struct B b;
    struct B *pb;
    struct A a = {10, 20};

    // Assigning addresses
    b.pa = &a;                       // b.pa requires a struct A type pointer
    b.i = 30;                        // b.i is an int type

    // Accessing via Structure Variable (b)
    b.pa->x // => 10
    b.pa->y // => 20
    b.i     // => 30

    /*
       b.pa->y <=> (*b.pa).y

       Operator Precedence for (*b.pa).y:
       Step ①: b.pa    (Evaluates the pointer member first)
       Step ②: *(b.pa) (Dereferences that pointer to reach the struct A instance)
       Step ③: .y      (Accesses the member y, yielding 20)
    */

    // ==========================================
    // 2. Chained Pointer Access
    // ==========================================
    pb = &b;

    pb->i       => 30
    pb->pa->x   => 10

    /*
       Equivalency Breakdown for pb->pa->x:
       Using the rule that x->y is functionally equivalent to (*x).y:

       The chained expression pb->pa->x can be expanded into these equivalent forms:
       1. (*pb).pa->x       (Dereferences pb first to get b, then uses -> on pa)
       2. (*pb->pa).x       (Dereferences pb->pa first to get a, then uses . on x)
       3. (*(*pb).pa).x     (Fully expanded explicit dereferencing for both pointers)
    */
}

/*
  Memory Layout Verification

     pb             b (struct B)             a (struct A)
    ╭───╮          ╭──────┬──────╮          ╭──────┬──────╮
    │500│ ───────> │ 100  │  30  │ ───────> │  10  │  20  │
    ╰───╯          ╰──────┴──────╯          ╰──────┴──────╯
     (500)          pa(500) i(502)           x(100) y(102)
*/
```

## Anonymous Structure

- Variables are created at the end of declaration only.

```c
// Anonymous vs. Named Structures

// ==========================================
// 1. Anonymous Structure (No Tag, No Variable)
// ==========================================
/*
   Defines a structure layout without a tag name.
   Because no variable is declared at the end, this specific definition
   cannot be used or instantiated elsewhere in the program.
*/
struct {
    int id;
    char name[36];
    int sal;
};                                // Anonymous structure


// ==========================================
// 2. Anonymous Structure with Immediate Variable
// ==========================================
/*
   Defines an anonymous structure and immediately instantiates a variable 'e1'.
   This creates a one-off grouped variable, but because there is no tag name,
   you cannot declare another variable of this exact type later in the code.
*/
struct {
    int id;
    char name[36];
    int sal;
} e1;


// ==========================================
// 3. Named Structure (Global Tag and Variables)
// ==========================================
/*
   Defines a structure with a tag name ('emp').
   This allows for immediate instantiation (like variable 'K') AND
   reusability later in the program.
*/
struct emp {
    int id;
    char name[36];
    int sal;
} K;

void main_structure_types() {
    // Because the structure is named 'emp', we can declare new instances anywhere
    struct emp e1;                // named structure
}
```

```c
#include <stdio.h>

struct emp {
    int id;
    char name[36];
    int sal;
};

void array_of_structures_and_pointers() {
    // Instead of individual variables (struct emp e1, e2, e3;)
    // We use an array of structures initialized inline:
    struct emp e[] = {
        {1, "abc", 3000},                // e[0]
        {2, "def", 4000},                // e[1]
        {3, "ghi", 5000}                 // e[2]
    };

    struct emp *p;
    p = e;                               // p points to base address of the array (e[0])

    /*
      ==========================================
      Memory Layout (Assuming 40 bytes per struct)
      ==========================================
         e[0]                 e[1]                 e[2]
      id   name   sal      id   name   sal      id   name   sal
      [ 1 | abc | 3000 ]   [ 2 | def | 4000 ]   [ 3 | ghi | 5000 ]
      500  502    538      540  542    578      580  582    618
       ▲
       │
     p(500)
    */

    // ==========================================
    // Precedence & Pointer Arithmetic Expressions
    // (Assuming each numbered expression is evaluated independently starting with p = &e[0])
    // ==========================================

    // ①
    (1)++p->sal
    // Evaluation: -> has higher precedence. Evaluates as ++(p->sal).
    // Result: Increments e[0].sal from 3000 to 3001.

    // ②
    (++p)->sal
    // Evaluation: Parentheses force pointer increment first. p moves to e[1] (540).
    // Result: 4000

    // ③
    ++p->name
    // Evaluation: Evaluates as ++(p->name). 'name' is an array base address (a constant pointer).
    // Result: C.E. (Compile Error: L-value required)

    // ④
    (++p)->name
    // Evaluation: Pointer increments to e[1], then accesses the name member.
    // Result: def

    // (5)
    e[p->id].name
    // Evaluation: p->id is 1. The expression becomes e[1].name.
    // Result: def

    // (6)
    e[--p->id].sal
    // Evaluation: -> has higher precedence. Evaluates as --(p->id). 1 becomes 0, resulting in e[0].sal.
    // Result: 3000

    // ==========================================
    // Additional Array & Pointer Evaluations
    // ==========================================

    e[1].id
    // Result: 2

    e[2].name[1]
    // Evaluation: Internally translates to *(e[2].name + 1).
    // Result: 'h'

    (*p)->name
    // Evaluation: *p yields a struct emp object (e[0]), not a pointer. The arrow operator -> requires a pointer.
    // Result: C.E. (Compile Error)

    *p->name
    // Evaluation: -> evaluates first. *(p->name) dereferences the first character of the string "abc".
    // Result: 'a'

    e->id
    // Evaluation: The array name 'e' decays to a pointer to its first element (&e[0]).
    // Result: 1
}
```

### Passing Structures to Functions (Pass-by-Reference)

Passing a structure to a function by reference (using pointers) is highly
recommended over passing by value, especially for large structures. Passing by
value creates a complete copy of the structure on the stack, which consumes
memory and processing time. Passing by reference only copies the pointer
(typically 2 to 4 bytes), allowing the function to modify the original structure
directly.

```c
// Passing Structures via Pointers

// Function expects a pointer to 'struct emp'
void test(struct emp *p) {
    // The function can now modify the original structure using p->member
}

void main_function_passing() {
    struct emp K;                       // Local structure instantiation

    // test(K);                         // X Incorrect: Passes the entire structure by value to a pointer parameter

    test(&K);                           // (ok) Correct: Passes the memory address of 'K' to the pointer parameter
}
```

## Nested Structures (Composition / Containership)

When a structure is completely defined inside another structure, it establishes
a relationship known as a nested structure, composition, or containership.

| Nesting Technique              | Syntax Characteristic                       | Outcome                                                                                                                                 |
| ------------------------------ | ------------------------------------------- | --------------------------------------------------------------------------------------------------------------------------------------- |
| **Definition Only**            | `struct emp { ... };` inside `struct org`   | Defines the template for `emp` within the scope of `org`, but does not allocate a member variable for it within the `org` memory block. |
| **Definition + Instantiation** | `struct emp { ... } e;` inside `struct org` | Defines the template and creates a member variable `e`, actively embedding the `emp` data into the `org` structure's memory footprint.  |

## Structure padding

```c
// Structure Padding, Memory Alignment, and Holes

#include <stdio.h>

// ==========================================
// 1. Default Memory Alignment (Padding & Holes)
// ==========================================
/*
   Compilers optimize memory access by aligning data on word boundaries (typically 4 or 8 bytes).
   To achieve this, the compiler automatically inserts empty spaces called "holes" (padding)
   between members.
*/
struct A_unoptimized {
    int i1;                             // 4 bytes
    char c1;                            // 1 byte
    int i2;                             // 4 bytes
    char c2;                            // 1 byte
};                                      // Theoretical size: 10 Bytes. Actual size: 16 Bytes.

/*
  Memory Layout with 16 Bytes (Padding inserted to align integers):

     i1 (4B)    c1(1B)  holes(3B)   i2 (4B)     c2(1B) holes(3B)
    ╭──────────┬──────┬───────────┬───────────┬──────┬───────────╮
    │          │      │           │           │      │           │
    ╰──────────┴──────┴───────────┴───────────┴──────┴───────────╯
    500        504    505         508         512
*/

// ==========================================
// 2. Manual Structure Packing (Member Reordering)
// ==========================================
/*
   By grouping smaller data types together, we can minimize the holes
   and reduce the overall memory footprint without disabling compiler alignment.
*/
struct A_optimized {
    char c1;                            // 1 byte
    char c2;                            // 1 byte
    int i1;                             // 4 bytes
    int i2;                             // 4 bytes
};                                      // Actual size reduced to: 12 Bytes.

/*
  Memory Layout with 12 Bytes Structure Packing:

    c1(1) c2(1) padding(2)  i1 (4B)       i2 (4B)
    ╭────┬────┬───────────┬─────────────┬─────────────╮
    │    │    │ 2 Bytes   │             │             │
    ╰────┴────┴───────────┴─────────────┴─────────────╯
    500  501              504           508
*/
```

## bit fields

- Reserving memory at bit level is called bit field

```c
// Bit Fields for Memory Optimization

#include <stdio.h>

// ==========================================
// 1. The Memory Waste Problem (Standard Structure)
// ==========================================
/*
   When storing state information with very few possible values, standard data types waste memory.
   For example, representing a sports match requires tracking four properties:

   1. match type: T20 (0), ODI (1), Test (2)
   2. match toss: head (0), tail (1)
   3. match status: Scheduled (0), ongoing (1), abonded (2), Completed (3), match stopped/Suspended (4)
   4. match result: WON (0), Loss (1), TIE (2), NR (3)
*/

// A standard structure using integers allocates full byte boundaries for each member.
struct match_standard {
    int types;
    int toss;
    int sts;
    int res;
};                                      // Total size: 8 Bytes (Assuming 16-bit integers)


// ==========================================
// 2. Bit Field Implementation
// ==========================================
/*
   By calculating the exact binary representation needed for the maximum value of each state,
   we can instruct the compiler to allocate memory at the bit level.

   Bit Requirements Calculation:
   - match type (max val 2 -> binary '10')   => Requires 2 bits
   - match toss (max val 1 -> binary '1')    => Requires 1 bit
   - match status (max val 4 -> binary '100')=> Requires 3 bits
   - match result (max val 3 -> binary '11') => Requires 2 bits
   Total bits required: 2 + 1 + 3 + 2 = 8 bits (Exactly 1 Byte)
*/

// The bit field syntax uses a colon ':' followed by the number of bits to allocate.
struct match {
    int type : 2;                       // Allocates exactly 2 bits
    int toss : 1;                       // Allocates exactly 1 bit
    int sts : 3;                        // Allocates exactly 3 bits
    int res : 2;                        // Allocates exactly 2 bits
};                                      // Total size: 1 Byte

void bit_field_usage() {
    struct match m1;

    // Values are assigned and accessed just like standard structure members.
    m1.type = 1;                        // Sets to ODI
    m1.toss = 0;                        // Sets to head
    m1.sts = 3;                         // Sets to Completed
    m1.res = 0;                         // Sets to WON

    // Note: Attempting to assign a value that exceeds the allocated bit width
    // (e.g., m1.toss = 2; which requires 2 bits) will result in overflow and data truncation.
}

```

### Memory analysis on bit field

```c
// Bit Field Signedness and Two's Complement Evaluation
#include <stdio.h>

// ==========================================
// 1. Default Signed Bit Fields and Overflow
// ==========================================
/*
   By default, bit fields declared as 'int' are signed.
   The Most Significant Bit (MSB) within the allocated bit width acts as the sign bit.
   Assigning a value whose binary representation triggers the sign bit will result
   in a negative number due to two's complement evaluation.
*/
struct A_default {
    int i : 3;                        // Allocates 3 bits. MSB is the sign bit.
    int j : 2;                        // Allocates 2 bits. MSB is the sign bit.
};                                    // Total size: 1 Byte

void default_signed_bitfields() {
    struct A_default a;
    a.i = 5;
    a.j = 1;

    // Output: -3 1
    printf("%d %d\n", a.i, a.j);

    /*
       Mathematical Breakdown for 'a.i = 5':
       Binary representation of 5 is '101'.
       Since 'i' is a 3-bit signed integer, the bit weights from left to right are:
       -2^2, 2^1, 2^0 (or -4, 2, 1).
       Calculation: 1(-4) + 0(2) + 1(1) = -3.

       Mathematical Breakdown for 'a.j = 1':
       Binary representation of 1 is '01'.
       For a 2-bit signed integer, the bit weights are: -2^1, 2^0 (or -2, 1).
       Calculation: 0(-2) + 1(1) = 1.
    */
}

// ==========================================
// 2. Explicit Signed vs. Unsigned Bit Fields
// ==========================================
/*
   To prevent unintended negative values when packing positive data, bit fields
   should be explicitly declared as 'unsigned'.
*/
struct A_explicit {
    signed i : 3;                     // 3-bit signed integer
    unsigned j : 3;                   // 3-bit unsigned integer
    signed k : 2;                     // 2-bit signed integer
    unsigned l : 2;                   // 2-bit unsigned integer
};                                    // Total size: 2 Bytes

void explicit_signedness_comparison() {
    struct A_explicit a;

    // Assigning the same integer value to both signed and unsigned 3-bit fields
    a.i = a.j = 5;

    // Assigning the same integer value to both signed and unsigned 2-bit fields
    a.k = a.l = 1;

    // Output: -3 5
    printf("%d %u\n", a.i, a.j);

    // Output: 1 1
    printf("%d %u\n", a.k, a.l);

    /*
       Comparison for 3-bit fields assigned '5' (Binary '101'):
       - 'i' (signed):   Weights are -4, 2, 1. Value = 1(-4) + 1(1) = -3.
       - 'j' (unsigned): Weights are 4, 2, 1.  Value = 1(4) + 1(1) = 5.

       Comparison for 2-bit fields assigned '1' (Binary '01'):
       - 'k' (signed):   Weights are -2, 1. Value = 0(-2) + 1(1) = 1.
       - 'l' (unsigned): Weights are 2, 1.  Value = 0(2) + 1(1) = 1.
    */
}
```

The bit-level representations demonstrate why strict type specification is
necessary when designing memory-constrained embedded systems or network packet
headers. Relying on default signed integers in bit fields can introduce subtle
arithmetic bugs when the data happens to toggle the most significant bit.
Transitioning to explicit `unsigned` declarations ensures standard binary
addition applies without two's complement interference.

### Practical Bit Field Optimization: Date Storage and Epoch Offsets

```c
#include <stdio.h>

// ==========================================
// 1. Standard Implementation (High Memory Cost)
// ==========================================
/*
   A standard structure allocating full integers for date components.
*/
struct date_standard {
    unsigned int day;
    unsigned int month;
    unsigned int year;
};                                          // Total size: 6 Bytes (Assuming 16-bit architecture)

// ==========================================
// 2. Partial Optimization
// ==========================================
/*
   Packing the day and month, but leaving the year as a standard integer.
*/
struct date_partial {
    unsigned d : 5;
    unsigned m : 4;
    unsigned y;
};                                          // Total size: 4 Bytes

// ==========================================
// 3. Full Optimization Using Epoch Offsets
// ==========================================
/*
   Bit Requirements Analysis:
   - day: values 1 to 31 -> Requires 5 bits
   - month: values 1 to 12 -> Requires 4 bits

   To aggressively pack the year, we establish an offset (epoch) starting at 1980.
   Allocating 7 bits provides a range of 0 to 127.
   This covers a time span from 1980 up to the year 2107.

   Total bits: 5 + 4 + 7 = 16 bits (Exactly 2 Bytes).
*/
struct date {
    unsigned d : 5;
    unsigned m : 4;
    unsigned y : 7;
};                                          // Total size: 2 Bytes

void date_offset_calculations() {
    struct date today;

    // ------------------------------------------
    // Encoding (Writing the year to memory)
    // ------------------------------------------
    // To store the year 2000, subtract the 1980 offset:
    // 2000 - 1980 = 20
    today.y = 20;

    // To store the year 2080:
    // 2080 - 1980 = 100
    today.y = 100;

    // ------------------------------------------
    // Decoding (Reading the year from memory)
    // ------------------------------------------
    // To retrieve the actual year, add the offset back:
    int current_year = today.y + 1980;      // If today.y is 100: 100 + 1980 = 2080

    /*
       Maximum Limits Demonstration:
       The highest value 7 bits can hold is 127.
       Max Year = 127 + 1980 = 2107
    */
}
```

## Enum

```c
// Enumerations (enum) for Named Constants and State Management

#include <stdio.h>

// ==========================================
// 1. Concept and Definition Rules
// ==========================================
/*
   An 'enum' is used to group named constants, effectively acting as an alias to 'int' in memory.
   This replaces raw integer codes with readable, self-documenting states.

   Common use cases include grouping network error codes:
   - Error 404 -> Page not found
   - Error 402 -> Server time out
   - Error 401 -> Connection lost
*/

// Defining an enumeration for a state machine, such as vehicle door statuses.
// By default, the compiler assigns a sequential integer sequence starting from 0.
enum doorstatus {
    LOCKED,                             // Automatically assigned 0 (binary 00)
    UNLOCKED,                           // Automatically assigned 1 (binary 01)
    OPEN,                               // Automatically assigned 2 (binary 10)
    CLOSED                              // Automatically assigned 3 (binary 11)
};

void enumeration_usage() {
    // ==========================================
    // 2. Instantiation and Assignment
    // ==========================================
    // Similar to structure instantiation (e.g., struct emp e1;), we declare an
    // enum variable using the 'enum' keyword followed by the tag and variable name.

    enum doorstatus d;

    // Assigning a named constant directly to the variable.
    // In hardware abstraction layers or RTOS development, this guarantees state
    // transitions use strict, readable conditions rather than arbitrary magic numbers.
    d = OPEN;

    // Output: 2
    // Because 'enum' evaluates as an alias to int, printing it with %d outputs its integer value.
    printf("%d", d);
}
```

### Enumeration Operations and Compile Errors

```c
#include <stdio.h>

enum doorstatus {
    LOCKED,                             // 0
    UNLOCKED,                           // 1
    OPEN,                               // 2
    CLOSED                              // 3
};                                      // -> named constants

void enum_operations() {
    enum doorstatus d1;                 // enum Variable (acts as int)

    d1 = OPEN;

    // 1. Standard Printing
    printf("%d", d1);                   // Output: 2

    // 2. Modifying an Enum Variable
    // Since 'd1' is a variable allocated in memory, it is a valid L-value
    // and can be incremented using arithmetic operators.
    printf("%d", ++d1);                 // (ok) Valid. Output: 3

    // 3. Modifying a Named Constant (Compile Error)
    // Named constants exist only as symbolic representations in the compiler,
    // not as physical memory locations.
    printf("%d", ++OPEN);            // X C.E: L-value required
}
```

### Advanced Enumeration Behaviors: Manual Values, Aliasing, and Limits

```c
#include <stdio.h>

// ==========================================
// 1. Manual Value Assignment and Progression
// ==========================================
/*
   While enumerators default to starting at 0, you can explicitly assign integer values.
   Any subsequent unassigned members will automatically increment by 1 from the preceding value.
*/
enum A {
    a,                          // 0 (Default starting value)
    b = 5,                      // 5 (Explicit assignment)
    c,                          // 6 (Increments from b)
    d = 3,                      // 3 (Explicit assignment)
    e = -7,                     // -7 (Explicit negative assignment)
    f,                          // -6 (Increments from e)
    g = 6                       // 6 (Explicit assignment)
};

// Practical Application: Setting a real-world base index
enum month {
    JAN = 1, FEB, MAR, APR, MAY, JUN, JUL, AUG, SEP, OCT, NOV, DEC
};    // The enumerator starts at 1, so DEC aligns correctly to 12.

// ==========================================
// 2. Data Type Constraints
// ==========================================
/*
   Enumerators must evaluate to valid integers.
*/
enum A_invalid {

       a_inv,
       b_inv,
       c_inv = 3.75,            // X C.E (Compile Error): float are not allowed.
       d_inv

};                              // A const integer expression Req (Required).

// ==========================================
// 3. Anonymous Enums and Typedef Aliasing
// ==========================================
/*
   Anonymous enums declare constants without creating a reusable tag name.
*/
enum { HEAD_anon, TAIL_anon };          // anonymous enum used purely for constant generation
enum { HEAD_inst, TAIL_inst } t1;       // Declares an anonymous enum and immediately instantiates variable 't1'

/*
   Combining 'typedef' with 'enum' generates robust custom data types,
   replacing raw integers with expressive domain logic.
*/
typedef enum toss { HEAD, TAIL } Toss;       // 'Toss' is now a custom data type
typedef enum boole { TRUE, FALSE } bool_t;   // 'bool_t' is now a custom boolean type

// ==========================================
// 4. Memory Footprint and Overflow Mechanics
// ==========================================
/*
   Because 'enum' is an alias to 'int', its memory allocation depends on the compiler
   and system architecture.

   Consequently, the integer ranges are:
   - 16-bit integer: -32768 to 32767
   - 32-bit integer: -2147483648 to 2147483647
*/
printf("%d", sizeof(enum test)); // Typically outputs 2 (or) 4 bytes

// Simulating overflow on a 16-bit architecture
enum A_overflow {
    a_max = 32767,              // Maximum positive value for a 16-bit signed integer
    b_over                      // Increments past the maximum
};

void enumeration_limits() {
    enum A_overflow k;
    k = b_over;

    // Output: -32768
    // The value wraps around to the lowest possible negative integer due to two's complement overflow.
    printf("%d", k);
}
```

## Union

```c
// Structure vs Union Memory Allocation

#include <stdio.h>

// ==========================================
// 1. The Core Difference: Memory Size
// ==========================================
/*
   The Difference between structure and union is size.
*/

// ------------------------------------------
// Structure Implementation
// ------------------------------------------
/*
   The size of structure is sum of all data members present inside the structure.
*/
struct emp_struct {
    int id;
    char name[36];
    int sal;
};                                      // 40 Bytes (Assuming 2-byte integers: 2 + 36 + 2)


// ------------------------------------------
// Union Implementation
// ------------------------------------------
/*
   The size of union is same as largest member present inside union
   Because all members in a union share the exact same starting memory address,
   the compiler only allocates enough contiguous memory to hold the single largest member.
*/
union emp_union {
    int id;
    char name[36];
    int sal;
};                                      // 36 Bytes (The 36-byte character array dictates the size)

void size_comparison() {
    // sum of all the members is size of struct
    // sizeof(struct emp): 40
    printf("Size of struct: %lu\n", sizeof(struct emp_struct));

    // size of union is same largest member present inside union
    // sizeof(union emp): 36
    printf("Size of union: %lu\n", sizeof(union emp_union));
}

```

### Memory analysis

```c
// Union Memory Sharing and Data Overwriting

#include <stdio.h>

// ==========================================
// 1. Shared Memory Space
// ==========================================
/*
   Unlike structures, all members of a union share the exact same starting memory address.
   Modifying one member directly overwrites the memory space accessed by the other members.
*/
union U {
    int i;
    char ch;
};                                      // Total size: 2 Bytes (dictated by the 2-byte integer)

void union_memory_overlap() {
    union U u;

    // ==========================================
    // 2. Integer Assignment and Byte-Level Extraction
    // ==========================================
    u.i = 515;

    // Output: 515 3
    printf("%d %d\n", u.i, u.ch);

    /*
       Memory Breakdown for u.i = 515 (Assuming Little-Endian Architecture):
       515 in 16-bit binary is: 0000 0010 0000 0011

       Memory addresses and contents:
       Address 501 (High Byte, bits 2^8 to ±2^15): 0000 0010 (Decimal 2)
       Address 500 (Low Byte, bits 2^0 to 2^7):    0000 0011 (Decimal 3)

       Because 'ch' is a 1-byte character, reading u.ch fetches only the lowest byte
       at address 500 (bits 2^0 to -2^7), which contains 0000 0011 (Decimal 3).
    */

    // ==========================================
    // 3. Partial Overwrite
    // ==========================================
    u.ch = 'A';                         // ASCII value of 'A' is 65 (Binary: 0100 0001)

    // Output: 577 65
    printf("%d %d\n", u.i, u.ch);

    /*
       Memory Breakdown after u.ch = 'A':
       Writing to 'ch' overwrites ONLY the lowest byte at address 500.

       Address 501 (High Byte): 0000 0010 (Unchanged from previous assignment)
       Address 500 (Low Byte):  0100 0001 (Overwritten with 65)

       When reading u.i, the processor combines both bytes again:
       0000 0010 0100 0001
       Calculation: (2 * 256) + 65 = 512 + 65 = 577.
    */
}
```

### Nested Structures within Unions and Advanced Memory Aliasing

```c
#include <stdio.h>

// ==========================================
// 1. Defining the Composite Data Types
// ==========================================
/*
   Assuming a 16-bit architecture where an 'int' occupies 2 bytes.
*/
struct A {
    int i;                              // 2 Bytes
    int j;                              // 2 Bytes
};                                      // Total size: 4 Bytes

struct B {
    int x;                              // 2 Bytes
    char ch[2];                         // 2 Bytes (1 Byte each)
};                                      // Total size: 4 Bytes

/*
   When structures are nested inside a union, the entire memory footprint of
   the structures overlaps completely. Both 'a' and 'b' start at the same
   base address and occupy the same 4 contiguous bytes.
*/
union U {
    struct A a;
    struct B b;
};                                      // Total union size: 4 Bytes (Dictated by the largest member)

void union_struct_aliasing() {          // (Renamed from main)
    union U u;

    // ==========================================
    // 2. Writing Data to the First Structure
    // ==========================================
    u.a.i = 512;
    u.a.j = 256;

    // ==========================================
    // 3. Reading Data through the Aliased Structure
    // ==========================================

    // Output: 512 0 1
    printf("%d %d %d\n", u.b.x, u.b.ch[0], u.b.ch[1]);

    /*
       ==========================================
       Memory Mapping Breakdown (Little-Endian)
       ==========================================
       The 4-byte memory block is shared simultaneously by struct A and struct B.

       Memory layout (Left to Right conceptually mapping High to Low bytes):

       struct A (a):
              <--------- j (256) --------> <--------- i (512) -------->
       Bits:  [0000 0001]    [0000 0000]   [0000 0010]    [0000 0000]

       struct B (b) Overlay:
              <--ch[1]-->    <--ch[0]-->   <----------- x ----------->
               (Value: 1)     (Value: 0)           (Value: 512)

       - 'u.b.x' perfectly overlaps 'u.a.i'. Since 'i' contains 512, 'x' reads exactly 512.
       - 'u.b.ch' overlaps 'u.a.j'. The integer 256 in little-endian format stores its
         low byte as 0x00 and its high byte as 0x01.
       - 'u.b.ch[0]' accesses the lower byte of 'j' (0000 0000), evaluating to 0.
       - 'u.b.ch[1]' accesses the higher byte of 'j' (0000 0001), evaluating to 1.
    */
}
```

### Practical example

```c
// Parsing a 16-Character Protocol (IMEI)

/*
   A 16-character IMEI number consists of specific sub-fields:
   - 2 characters: CountryCode
   - 2 characters: MFC
   - 2 characters: SAR
   - 4 characters: MODEL NUMBER
   - 6 characters: UNQ SERIAL NUMB
*/

// Memory Aliasing with Unions (Type Punning)

#include <stdio.h>
#include <string.h>

// ==========================================
// 1. The Standard Structure Approach
// ==========================================
struct IMEI {
    char cc[2];
    char mfc[2];
    char sar[2];
    char mn[4];
    char usn[6];
};                                      // Total Size: 16 Bytes

void standard_struct_parsing() {
    struct IMEI i;

    // Writing to individual structured fields
    strcpy(i.cc, "10");
    strcpy(i.mfc, "35");
    strcpy(i.sar, "12");
    strcpy(i.mn, "1769");
    strcpy(i.usn, "1AB341");

    // Reading requires formatting each segment individually
    // Output: 10351217691AB341
    printf("%s%s%s%s%s", i.cc, i.mfc, i.sar, i.mn, i.usn);
}

// ==========================================
// 2. The Union Aliasing Approach
// ==========================================
/*
   By wrapping the structure inside a union alongside a flat character array,
   we create two simultaneous views of the exact same 16-byte memory block.
*/
union OIMEI {
    struct IMEI in;                     // Structured view (16 Bytes)
    char obuffer[16];                   // Flat buffer view (16 Bytes)
};                                      // Total union size: 16 Bytes

void union_aliasing_parsing() {
    union OIMEI o;

    // Writing data using the structured view
    strcpy(o.in.cc, "10");
    strcpy(o.in.mfc, "35");
    strcpy(o.in.sar, "12");
    strcpy(o.in.mn, "1769");
    strcpy(o.in.usn, "1AB341");

    // Reading the entire packet at once using the flat buffer view
    // The union allows us to print the contiguous 16 bytes directly without assembling parts.
    printf("%s", o.obuffer);
}
```
