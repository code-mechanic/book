# String

1. Array of characters is called character Array (or) String (or) String
   Constant
2. Any content which is enclosed in pair of double Quotes is called string
3. Every string is Replaced with starting Address of string.
4. Every string is terminated with NULL ('\0') character.

## Introduction to string

- Character Representation & Memory Allocation
  - **Standard ASCII & Bit Requirement:** Standard ASCII defines **128
    characters**, which mathematically requires **7 bits**.
  - **Byte Alignment:** Because C lacks a native 7-bit data type, memory is
    allocated in **8 bits (1 byte)**.
  - **Combinations:** An 8-bit memory space provides **256 possible value
    combinations** (0 to 255).

- signed char (Standard ASCII Set)
  - **Range:** **-128 to 127**.
  - **Standard ASCII (0 to 127):** Stores standard alphabets, digits, and basic
    symbols.
  - **Negative Values (-128 to -1):** Reserved for special system
    functionalities rather than printable characters—most notably **`EOF` (End
    of File)**.

- unsigned char (Extended ASCII Set)
  - **Range:** **0 to 255**.
  - **Extended ASCII (128 to 255):** Created to support additional mathematical
    and technical symbols—such as the degree symbol (`°`, ASCII value 248),
    square root (`√`), Sigma (`Σ`), and integration (`∫`).
  - **Usage:** Any character code above 127 cannot fit in a `signed char` and
    strictly requires an **`unsigned char`**.

```c
#include <stdio.h>

void format_specifier_edge_cases() {
    // ---------------------------------------------------------
    // Quote and Backslash Escapes
    // ---------------------------------------------------------

    // 11) C.E : double quote cancels each other when 2 are detected
    /* printf("" hello ""); */

    // 12) Valid string concatenation via adjacent quotes
    printf("" "" "hello" "" "");        // Output: hello

    // 13) Invalid syntax due to unmatched/canceling quotes
    /* printf("" "" "" hello "" "" ""); */ // C.E

    /* printf("\"); */                  // C.E

    // 16) Escaping a backslash
    printf("\\");                       // Output: \

    // 17) New line
    printf("\n");                       // Output: new line ↓

    // 18) Escaped backslash followed by 'n'
    printf("\\n");                      // Output: \n

    // 19) Escaped backslash followed by new line
    printf("\\\n");                     // Output: \ ↓ (backslash then new line)

    // 20) Two escaped backslashes followed by 'n'
    printf("\\\\n");                    // Output: \\n


    // ---------------------------------------------------------
    // Percent Sign (%) and Format Specifier Edge Cases
    // ---------------------------------------------------------

    // 21) Malformed percent specifier (single %)
    printf("%d % %d = %d", 5, 2, 5%2);  // Output: 5 % d = 2

    // 22) Correct escaping of percent sign (%%)
    printf("%d %% %d = %d", 5, 2, 5%2); // Output: 5 % 2 = 1

    // 23) Single percent sign
    printf("%");                        // tc : %   | gcc : no o/p

    // 24) Double percent sign
    printf("%%");                       // tc : %   | gcc : %

    // 25) Triple percent sign
    printf("%%%");                      // tc : %%  | gcc : %

    // 26) Quadruple percent sign
    printf("%%%%");                     // tc : %%  | gcc : %%

    // 27) Standard integer format
    printf("%d", 10);                   // Output: 10

    // 28) Missing argument for format specifier
    printf("%d");                       // Output: G.V (Garbage Value)

    // 29) Escaped percent sign before 'd'
    printf("%%d");                      // Output: %d

    // 30) Escaped percent sign ignoring the argument
    printf("%%d", 10);                  // Output: %d

    // 31) Escaped percent sign followed by valid format specifier
    printf("%%%d", 10);                 // Output: %10

    // 32) Escaped percent sign followed by format specifier missing an argument
    printf("%%%d");                     // Output: % G.V (Garbage Value)
}
```

## 1D Strings

- Array of character is called character array (or) string (or) string constant.
- Any content which is represent in pair of Double Quote is called "string
  constant".
- Every string is by default terminal with NULL character.
- Every string is replaced with base address of string (or) Starting address.

```c
// String Format Specifiers, Null Terminators, and Array Decay

#include <stdio.h>

void string_manipulation_analysis() {
    // Array initialization with explicit null terminator
    char ch[] = {'a', 'b', 'c', 'd', 'e', 'f', '\0'};

    // Normal string print. '%s' expects a char* (base address) and prints until '\0'
    printf("%s", ch);        // Output: abcdef

    // Printing the decayed array pointer as an integer (yields base address)
    printf("%d", ch);        // Output: 500

    // Injecting a null terminator artificially truncates the string
    ch[3] = '\0';
    printf("%s", ch);        // Output: abc

    // Overwriting the artificial null terminator with ASCII 100 ('d')
    ch[3] = 100;
    printf("%s", ch);        // Output: abcdef

    // Overwriting the legitimate null terminator at the end
    ch[6] = 'g';
    printf("%s", ch);        // Output: abcdefgGCGCGC..... (Garbage values until random '\0')

    // Restoring the null terminator (0 is equivalent to '\0')
    ch[6] = 0;

    // Pointer arithmetic to print a substring starting at index 3
    printf("%s", &ch[3]);    // Output: def

    // Attempting to increment the array name directly
    printf("%s", ++ch);   // C.E: L-value required (ch is a constant pointer)
}

/*
        ch[0]    1      2      3      4      5      6
      ╭──────┬──────┬──────┬───────┬──────┬──────┬──────╮
      │  'a' │  'b' │  'c' │ ~'d'~ │  'e' │  'f' │~'\0'~│
      │      │      │      │ ~'\0'~│      │      │ ~'g'~│
      │      │      │      │  'd'  │      │      │  '\0'│
      ╰──────┴──────┴──────┴───────┴──────┴──────┴──────╯
        500    501    502    503    504    505    506
*/
```

- **Format Specifiers:**
  - `%d` $\Rightarrow$ `int`
  - `%c` $\Rightarrow$ `char`
  - `%f` $\Rightarrow$ `float`
  - `%s` $\Rightarrow$ ~~`string`~~ (Conceptually incorrect to strictly call it
    a primitive string type which is not supported by C)
  - `%s` $\Rightarrow$ `char*` (Accurate definition: expects a character
    pointer)

- **Address Evaluation:**
  - `%s` $\Rightarrow$ `char*`
  - `ch` $\Rightarrow$ `500` $\Rightarrow$ `char*`
  - `&ch[3]` $\Rightarrow$ `&*(ch+3)` $\Rightarrow$ `ch+3` (Evaluates to address
    `503`)

## Pointers to string

### Character Array vs Character Pointer

| Feature                    | Character Array                                                                    | Character Pointer                                                                                      |
| -------------------------- | ---------------------------------------------------------------------------------- | ------------------------------------------------------------------------------------------------------ |
| **Storage & Purpose**      | `s1` is an array which is used to store the string. Allocates 100 bytes of memory. | `s1` is a pointer pointing to the base address of the string. Allocates 6 bytes for the string itself. |
| **Concatenation Behavior** | If we perform concatenation, the result is well defined.                           | If we perform concatenation, the result is not well defined.                                           |
| **Pointer Nature**         | `s1` is a constant pointer.                                                        | `s1` is a normal pointer, not a constant pointer.                                                      |
| **Address Modification**   | `++s1` results in a Compile Error (C.E).                                           | `++s1` is valid and does not result in a Compile Error.                                                |
| **Element Access**         | `s1[2]` is equivalent to `*(s1+2)`.                                                | `*(s1+2)` is equivalent to `s1[2]`, which yields `'l'`.                                                |

- Program

```c
void compare_string_initialization() {

    // ==========================================
    // 1. Character Array
    // ==========================================
    char s1_arr[100] = "Hellow";  // 100 bytes allocated

    // Element access equivalence
    char c1 = s1_arr[2];          // Equivalent to *(s1_arr + 2)

    // Incrementing the array name
    ++s1_arr;                  // -> C.E (Compile Error) because s1_arr is a constant pointer

    // ==========================================
    // 2. Character Pointer
    // ==========================================
    char *s1_ptr = "Hellow";      // Pointer to base address of 6-byte string

    // Element access equivalence
    char c2 = *(s1_ptr + 2);      // Equivalent to s1_ptr[2] -> 'l'

    // Incrementing the pointer
    ++s1_ptr;                     // -> Valid (Not C.E) because s1_ptr is a normal pointer

    // Concatenation: Not well defined since it points to a read-only literal with no extra space
}

```

### Array of Pointers to Strings and Pointer Arithmetic

```c
void array_of_string_pointers() {
    // Individual pointer initialization
    char *s1 = "apple";
    char *s2 = "mango";
    char *s3 = "orange";

    // Equivalent array of pointers initialization
    char *s[] = {"apple", "mango", "orange"};

    // Size evaluation (Assuming 2-byte pointers for a 16-bit architecture)
    sizeof(s) ; 6
}

/*
        s[0]       s[1]       s[2]
     ╭──────────┬──────────┬──────────╮
     │   500    │   600    │   700    │
     ╰────┬─────┴────┬─────┴────┬─────╯
         100        102        104
          │          │          │
          ▼          ▼          ▼
        apple      mango      orange
       (500)      (600)      (700)
*/
```

- **Pointer Arithmetic and Dereferencing Evaluations**
  - Based on the base addresses `s = 100`, `"apple" = 500`, `"mango" = 600`, and
    `"orange" = 700`:
  - **Base Type & Address:** `s => 100 => char **`
  - **First String Address:** `*s => 500` (Equivalent to `s[0]`). If printed
    with `%s`, evaluates to `apple`.
  - **First Character:** `**s` evaluates to `'a'` (the first character of
    "apple").
  - **2D Array Indexing Equivalent:** `s[1][3]` is equivalent to `*(*(s+1)+3)`.
    This evaluates to `'g'` (the 4th character of "mango").
  - **Pointer Offset String:** `s[2] + 1` evaluates to `range` (Advances the
    base address of "orange" by 1 byte).
  - **Dereferencing Array Element:** `*s[1]` evaluates to `'m'` (the first
    character of "mango").

## 2D Character Arrays

```c
#include <stdio.h>

// ==========================================
// 2D Character Array Initialization & Memory Layout
// ==========================================
void two_dimensional_string_arrays() {

    char S[][] = {"Hyderabad", "Pune", "Delhi", "Chennai"}; // Wrong (Column size mandatory)


    // Correct: Column size (10) dictates the fixed byte boundary per string
    char S[][10] = {{"Hyderabad"}, {"Pune"}, {"Delhi"}, {"Chennai"}};
    char S[][10] = {"Hyderabad", "Pune", "Delhi", "Chennai"};

    /*
       Memory Layout for S[][10] (Assuming base address 500):
       ╭─────────────┬─────────────┬─────────────┬─────────────╮
       │ Hyderabad\0 │ Pune\0      │ Delhi\0     │ Chennai\0   │
       ╰─────────────┴─────────────┴─────────────┴─────────────╯
        500           510           520           530
    */

    /*
       Explicit Initialization Equivalents:
       This is the SAME as the string literal initialization above:
    */
    char S[][10] = { {'h','y','d','e','r','a','b','a','d','\0'},
                     {'p','u','n','e','\0'} };

    /*
       This is DIFFERENT (Lacks explicit row boundaries via nesting):
       Here, no row boundary is there, so compiler consecutively allocates memory for each character.
    */
    char S[][10] = { 'h','y','d','e','r','a','b','a','d','\0', 'p','u','n','e','\0', ... };


    // --- 2D Array Pointer Arithmetic Evaluations ---
    // Expression => Evaluated Address/Value => Format Specifier => Output

    // String Outputs (%s) - Single level of indirection
    *(S+2)+2      // => 520+2 => 522       => %s => lhi
    S[1]+2        // => 510+2 => 512       => %s => ne
    *S+3          // => *500+3 => 503      => %s => erabad
    *(S+3)        // => Base of 4th string => %s => Chennai

    // Character Outputs (%c) & ASCII Arithmetic - Double level of indirection
    *(*(S+3)+1)   // => *531              => %c => h
    *(*(S+1)+2)+3 // => *512+3 => 'n'+3   => %c => q (ASCII arithmetic)
    *S[2]         // => First char        => %c => d
    *S[3]+3       // => 'c' + 3           => %c => f
    S[1][1]       // => Array subscript   => %c => u
    S[2][3] + 3   // => 'h' + 3           => %c => k
}
```

```c
    if (s1 == s2) {
```

### strcpy

```c
// String Copy Behaviors and Custom Implementation

#include <stdio.h>
#include <string.h>

// ==========================================
// 1. Standard strcpy Behaviors
// ==========================================
void standard_strcpy_examples() {
    char s1[5] = "hai";
    char s2[5];

    // s2 = s1;                   // Invalid (L-value required)
    strcpy(s2, s1);               // Valid string copy
    printf("%s", s2);             // Output: hai

    char s3[10];

    strcpy(s3, "world");
    // Output conceptually: world

    strcpy(s3, "india" + 2);      // Pointer arithmetic offset on string literal
    // Output conceptually: dia

    strcpy(s3, "hello\0india");   // Copies until the first null terminator
    // Output conceptually: hello
}

/*
  Memory Layout for strcpy(s2, s1)

     s1  ╭───┬───┬───┬────┬───╮
         │ h │ a │ i │ \0 │ 0 │
         ╰───┴───┴───┴────┴───╯
          500 501 502  503  504

     s2  ╭───┬───┬───┬────┬───╮
         │ h │ a │ i │ \0 │   │
         ╰───┴───┴───┴────┴───╯
          600 601 602  603  604
*/


// ==========================================
// 2. Custom String Copy Implementation
// ==========================================

/*
  Progressive optimization of a custom string copy loop:

  Version 1 (Verbose):
  while(*s) {
      *t = *s;
      ++t;
      ++s;
  }
  *t = *s;

  Version 2 (Post-increment):
  while(*s) {
      *t++ = *s++;
  }
  *t = *s;

  Version 3 (Fully optimized):
  while(*t++ = *s++);
*/

char* custom_strcopy(char *t, char *s) {
    char *temp = t;                         // Store base address to return

    // Fully optimized loop:
    // Copies character, increments pointers, and evaluates the assigned value.
    // Stops when '\0' (evaluates to 0/false) is copied.
    while (*t++ = *s++);

    return temp;
}
```

```c
  Memory Layout for strcat(s1, s2)
  The null terminator of the destination string is overwritten by the first character of the source string.

          500             (504)

     s2  ╭───┬───┬───┬───┬────╮
         │ b │ a │ l │ l │ \0 │
         ╰───┴───┴───┴───┴────╯

// ==========================================
// 2. Custom String Concatenation Implementation
// ==========================================

void custom_strconcat(char *t, char *s) {
    // Step 1: Traverse the destination string until the null terminator is reached
    while (*t) {
        ++t;
    }

    // Step 2: Copy the source string into the destination starting at the null terminator
    while (*t++ = *s++);
}

    char s2[10] = "book";

    custom_strconcat(s1, s2);             // Modifies s1 to "notebook"
```

```c
    strlen("hello" + 2);               // Evaluates to: 3

    strlen("hello\0india");            // Evaluates to: 5 (stops counting at the first '\0')

    }
```

```c
// String Comparison and Custom Implementation

#include <stdio.h>
#include <string.h>

// ==========================================
// 1. Standard strcmp Behaviors & Return Values
// ==========================================
void standard_strcmp_evaluations() {
    /*
       strcmp compares strings character by character using their ASCII values.
       It returns the difference of the first non-matching character pair.
    */

    // Example 1: "ball" vs "bell"
    // Mismatch at index 1: 'a' (97) - 'e' (101) = -4
    strcmp("ball", "bell");        // Evaluates to: -4

    // Example 2: "hello" vs "hai"
    // Mismatch at index 1: 'e' (101) - 'a' (97) = 4
    strcmp("hello", "hai");        // Evaluates to: 4

    // Example 3: Different lengths, first string longer
    // "rama" vs "ram\0"
    // Mismatch at index 3: 'a' (97) - '\0' (0) = 97
    strcmp("rama", "ram");         // Evaluates to: 97

    // Example 4: Different lengths, second string longer
    // "ram\0" vs "rama"
    // Mismatch at index 3: '\0' (0) - 'a' (97) = -97
    strcmp("ram", "rama");         // Evaluates to: -97

    // Example 5: Identical strings
    strcmp("ram", "ram");          // Evaluates to: 0

    // Example 6: Case sensitive comparison
    // "Ball" vs "bell"
    // Mismatch at index 0: 'B' (66) - 'b' (98) = -32
    strcmp("Ball", "bell");        // Evaluates to: -32

    // Example 7: Case insensitive comparison (stricmp - compiler specific)
    stricmp("Ball", "ball");       // Evaluates to: 0 (ignore case)
}

/*
  strcmp Return Value Interpretation:
  -----------------------------------
  strcmp(s1, s2);
  +ve : s1 should come after s2 (in dictionary order)
  -ve : s1 should come before s2 (in dictionary order)
   0  : s1 & s2 are equal
*/


// ==========================================
// 2. Custom String Comparison Implementation
// ==========================================

int strcompare(char *t, char *s) {
    // Loop continues as long as we haven't reached the end of 't' (*t is not '\0')
    // AND the characters at the current positions of 't' and 's' match (*t == *s).
    while (*t && *t == *s) {
        ++t;
        ++s;
    }

    // Return the ASCII difference of the first mismatching characters (or 0 if both hit '\0')
    return *t - *s;
}

void main_strcompare() {
    char s1[] = "rama";
    char s2[] = "ranga";
    int diff;

    // Mismatch at index 2: 'm' (109) - 'n' (110) = -1
    diff = strcompare(s1, s2);
    // diff will be -1
}
```

```c
// ==========================================
// 1. printf Format Specifiers vs Direct Array Passing
// ==========================================
void printf_format_security() {
    char s1[] = "hello";

    // These all produce the same output ("hello")
    printf("%s", s1);
    printf("%s", "hello");
    printf("hello");
    printf(s1);

    // However, consider this scenario:
    char s2[] = "hello %d world %d";

    // Recommended approach:
    printf("%s", s2);                              // Output: hello %d world %d

    // Not recommended (Security/Formatting risk):
    printf(s2);                                    // Output: hello G.v world G.v (Garbage values)

    /*
      Why the difference?
      In printf, the 1st argument is meant for format specifying and printing.
      The 2nd and so on are meant for substitution for format specifiers only, not printing.
      If you pass s2 directly, printf interprets '%d' as format specifiers and looks for
      subsequent arguments that do not exist, printing garbage.
    */
}

// 2. Advanced printf Pointer Arithmetic
// ==========================================
void printf_pointer_arithmetic() {
    char s1[] = "hello";

    // Shifting base addresses
    printf("%s", s1 + 2);                          // Output: llo
    printf("%s", "hello" + 2);                     // Output: llo

    // Dereferencing to single characters
    printf("%c", *("hello" + 2));                  // Output: l
    printf("%s", &"hello"[2]);                     // Output: llo
    printf("%c", *"hello");                        // Output: h

    // Embedded nulls truncate the output
    printf("%s", "hello\0India");                  // Output: hello
    printf("ice\0cream");                          // Output: ice

    // Arithmetic evaluation inside format strings
    // 'b' - 'b' = 0, so "hello" + 0 remains base address 500
    printf("hello" + 'b' - 'b');                   // Output: hello

    printf("hello %d" + 5, 10);                    // Output:  10

    printf("th %s %c%c easy", "this" + 2, *"in", 2["was"]); // Output: th is is easy
```
